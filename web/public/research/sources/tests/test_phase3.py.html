  10     def test_three_hand_enumerated_competitions(self):
  11         # With no day-2 trucks and at most one on day 1, these are all choices.
  12         # Chips: <=6 GJ over two days, or 4 + (2 - 4/3)*2 GJ without storage.
  13         # Briquettes: 6 GJ in either case. Pellets need 0.5 m3 closing on day 1.
  14         candidates = {
  15             "base": {"gas": 24, "wc": 8.4, "wp": 5, "bq": 10},
  16             "tight-silo": {"gas": 24, "wc": 8.4, "bq": 10},
  17             "no-storage": {"gas": 24, "wc": 10.4, "bq": 10},
  18         }
  19         for case, costs in candidates.items():
  20             with self.subTest(case=case):
  21                 scenario = synthetic_scenario(case)
  22                 result = solve(scenario)
  23                 winner = min(costs, key=costs.__getitem__)
  24                 self.assertEqual(result.status, 0)
  25                 self.assertAlmostEqual(result.best_bound, costs[winner])
  26                 self.assertAlmostEqual(result.mip_gap, 0)
  27                 self.assertAlmostEqual(sum(day.cost for day in result.audit), costs[winner])
  28                 self.assertEqual(result.plans[0].trucks, {f: int(f == winner) for f in scenario.feedstocks})
  29                 self.assertEqual(sum(result.plans[1].trucks.values()), 0)

  54     def test_two_fuels_delivered_and_burned_together(self):
  55         scenario = synthetic_scenario()
  56         first = replace(scenario.days[0], truck_limit=2, total_storage_m3=2, gas_capacity_gj=0)
  57         # Each required closing stock forces its own truck. Both trucks also
  58         # leave 1 m3 to burn. Chips must supply 3 GJ in the new unit; the
  59         # briquettes split 2 GJ existing and 1 GJ new to meet demand of 6.
  60         scenario = replace(
  61             scenario, days=(first,), terminal_m3={"wc": 1, "wp": 0, "bq": 1},
  62         )
  63         result = solve(scenario)
  64         self.assertEqual(result.status, 0)
  65         self.assertEqual(result.plans[0].trucks, {"wc": 1, "wp": 0, "bq": 1})
  66         self.assertAlmostEqual(result.best_bound, 6.4)
  67         self.assertAlmostEqual(result.mip_gap, 0)
  68         self.assertAlmostEqual(result.audit[0].biomass_gj_by_plant_and_fuel["new"]["wc"], 3)
  69         self.assertAlmostEqual(result.audit[0].biomass_gj_by_plant_and_fuel["existing"]["bq"], 2)
  70         self.assertAlmostEqual(result.audit[0].biomass_gj_by_plant_and_fuel["new"]["bq"], 1)
  71         self.assertEqual(solve(replace(scenario, days=(replace(first, truck_limit=1),))).status, 2)
  72         self.assertEqual(solve(replace(scenario, days=(replace(first, total_storage_m3=1.9),))).status, 2)

  82     def test_total_storage_is_a_shared_limit_and_silo_is_pellet_only(self):
  83         scenario = synthetic_scenario()
  84         first = replace(scenario.days[0], truck_limit=2, total_storage_m3=4, pellet_silo_m3=3)
  85         plan = DayPlan({"wc": 1, "wp": 1, "bq": 0}, {"wc": 2, "wp": 2.5, "bq": 0},
  86                        {p: {f: Burn(0, 0) for f in scenario.feedstocks} for p in scenario.plants}, 6)
  87         new = replace(scenario, days=(first,), terminal_m3=None)
  88         with self.assertRaisesRegex(ValueError, "total storage"):

  91             verify_plan(replace(new, days=(replace(first, total_storage_m3=5, pellet_silo_m3=1),)), [plan])

  93         verify_plan(replace(new, days=(replace(first, pellet_silo_m3=0),)), [nonpellets])

  95     def test_cross_plant_fuel_availability_checked_by_phase_one(self):
  96         scenario = synthetic_scenario()
  97         result = solve(scenario)
  98         two_plants = result.plans[0]
  99         # Keep the same plant heat and inventory arithmetic but misclassify a
 100         # fresh draw as stored even though the day started empty.
 101         burns = {plant: dict(fuels) for plant, fuels in two_plants.burns.items()}
 102         burns["new"]["wp"] = Burn(0.5, 0.5)
 103         mutated = replace(two_plants, burns=burns)
 104         with self.assertRaisesRegex(ValueError, "stored burn exceeds opening"):
 105             verify_plan(scenario, (mutated, result.plans[1]))
 106 
 107         second = result.plans[1]
 108         burns = {plant: dict(fuels) for plant, fuels in second.burns.items()}
 109         burns["new"]["wp"] = Burn(0.5, 0)
 110         with self.assertRaisesRegex(ValueError, "fresh burn exceeds deliveries"):
 111             verify_plan(scenario, (two_plants, replace(second, burns=burns)))
