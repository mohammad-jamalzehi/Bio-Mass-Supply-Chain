  10 class SolverValidationTests(unittest.TestCase):
  11     def setUp(self):

  14     def test_optimum_matches_hand_enumeration_and_solver_bound(self):
  15         # Separate oracle for this fixture: every day-3 truck delivers 9 GJ
  16         # against demand of only 3 GJ and zero required closing stock.
  17         costs = {}
  18         for first, third in product((0, 1), repeat=2):
  19             if third == 0:
  20                 costs[first, 0, third] = 6 * first + 3 * (18 - 9 * first)
  21         self.assertEqual(costs, {(0, 0, 0): 54, (1, 0, 0): 33})
  22         best_schedule = min(costs, key=costs.__getitem__)
  23 
  24         result = solve(self.scenario)
  25         self.assertEqual(result.status, 0)
  26         self.assertEqual(tuple(day.trucks for day in result.days), best_schedule)
  27         self.assertAlmostEqual(result.totals.cost, costs[best_schedule])
  28         self.assertAlmostEqual(result.best_bound, costs[best_schedule])
  29         self.assertAlmostEqual(result.mip_gap, 0)
  30         self.assertAlmostEqual(result.totals.emissions_tco2e, 5.4)
  31         self.assertEqual(
  32             [(d.inventory.opening, d.inventory.delivered, d.inventory.fresh_burn,
  33               d.inventory.stored_burn, d.inventory.closing, d.gas_heat_gj) for d in result.days],
  34             [(0, 3, 2, 0, 1, 0), (1, 0, 0, 1, 0, 6), (0, 0, 0, 0, 0, 3)],
  35         )
  36         validate_inventory((day.inventory for day in result.days), initial=0, terminal=0)

  38     def test_tighter_plant_capacity_makes_day_one_truck_unusable(self):
  39         days = (replace(self.scenario.days[0], plant_capacity_gj=3), *self.scenario.days[1:])
  40         result = solve(replace(self.scenario, days=days))
  41         self.assertEqual(result.status, 0)
  42         self.assertAlmostEqual(result.totals.cost, 54)
  43         self.assertEqual(tuple(day.trucks for day in result.days), (0, 0, 0))

  45     def test_tighter_storage_makes_day_one_truck_unusable(self):

  51     def test_insufficient_heat_capacity_is_infeasible(self):
  52         days = (self.scenario.days[0], replace(self.scenario.days[1], gas_capacity_gj=5),
  53                 self.scenario.days[2])
  54         result = solve(replace(self.scenario, days=days))
  55         self.assertEqual(result.status, 2)
  56         self.assertIsNone(result.totals)
  57         self.assertEqual(result.days, ())

  59     def test_opening_stock_available_on_first_day_and_terminal_target_optional(self):
  60         scenario = replace(
  61             self.scenario, days=(DayLimits(3, 0, 1, 3, gas_capacity_gj=0),),
  62             initial_inventory_m3=2, terminal_inventory_m3=None,
  63         )
  64         result = solve(scenario)
  65         self.assertEqual(result.status, 0)
  66         self.assertEqual(result.days[0].inventory.stored_burn, 1)
  67         self.assertEqual(result.days[0].inventory.closing, 1)
  68         self.assertEqual(solve(replace(scenario, terminal_inventory_m3=0)).status, 2)

  77     def test_audit_rejects_corrupted_heat_and_cost(self):
