  50 @dataclass(frozen=True)
  51 class Scenario:
  52     name: str
  53     feedstocks: Mapping[str, Feedstock]
  54     plants: Mapping[str, Plant]
  55     days: tuple[Limits, ...]
  56     initial_m3: Mapping[str, float]
  57     terminal_m3: Mapping[str, float] | None
  58     gas_cost_per_useful_gj: float
  59     gas_emissions_per_useful_gj: float
  60 
  61     def __post_init__(self) -> None:
  62         fuels, plants = set(self.feedstocks), set(self.plants)
  63         if not fuels or not plants or not self.days:
  64             raise ValueError("feedstocks, plants and days are required")
  65         for fuel in self.feedstocks.values():
  66             fuel.heat(1, 1)
  67             nonnegative("truck volume", fuel.truck_m3)
  68             nonnegative("biomass price", fuel.cost_per_dry_tonne)
  69             if fuel.truck_m3 == 0:
  70                 raise ValueError("truck volume must be positive")
  71         for plant in self.plants.values():
  72             if set(plant.efficiency) != fuels or set(plant.emissions_per_useful_gj) != fuels:
  73                 raise ValueError("plant efficiencies and emissions factors must cover all feedstocks")
  74             for key, fuel in self.feedstocks.items():
  75                 fuel.heat(1, plant.efficiency[key])
  76                 nonnegative("biomass emissions factor", plant.emissions_per_useful_gj[key])
  77         for stocks in (self.initial_m3, self.terminal_m3):
  78             if stocks is not None:
  79                 if set(stocks) != fuels:
  80                     raise ValueError("inventory targets must cover all feedstocks")
  81                 for value in stocks.values():
  82                     nonnegative("inventory target", value)
  83         nonnegative("gas price", self.gas_cost_per_useful_gj)
  84         nonnegative("gas emissions factor", self.gas_emissions_per_useful_gj)
  85         for day in self.days:
  86             if type(day.truck_limit) is not int or day.truck_limit < 0:
  87                 raise ValueError("truck limit must be a nonnegative integer")
  88             if set(day.plant_heat_capacity_gj) != plants:
  89                 raise ValueError("daily limits must cover all plants")
  90             for value in (day.demand_gj, day.total_storage_m3, *day.plant_heat_capacity_gj.values()):
  91                 nonnegative("daily limit", value)
  92             for optional in (day.pellet_silo_m3, day.gas_capacity_gj):
  93                 if optional is not None:
  94                     nonnegative("optional daily limit", optional)
  95             if day.pellet_silo_m3 is not None and "wp" not in fuels:
  96                 raise ValueError("a pellet silo requires the wood-pellet feedstock wp")

 131 def verify_plan(scenario: Scenario, plans: Sequence[DayPlan]) -> tuple[DayAudit, ...]:
 132     """Recalculate physical and monetary totals independently of the solver matrix."""
 133     fuels, plants = scenario.feedstocks, scenario.plants
 134     opening = dict(scenario.initial_m3)
 135     if len(plans) != len(scenario.days):
 136         raise ValueError("expected one plan per day")
 137     audited = []
 138     for number, (plan, limits) in enumerate(zip(plans, scenario.days, strict=True), start=1):
 139         if set(plan.trucks) != set(fuels) or set(plan.closing_m3) != set(fuels):
 140             raise ValueError(f"day {number}: incomplete fuel plan")
 141         if set(plan.burns) != set(plants) or any(set(burns) != set(fuels) for burns in plan.burns.values()):
 142             raise ValueError(f"day {number}: incomplete plant allocation")
 143         if any(type(n) is not int or n < 0 for n in plan.trucks.values()) or sum(plan.trucks.values()) > limits.truck_limit:
 144             raise ValueError(f"day {number}: truck counts violate integrality or shared limit")
 145 
 146         inventory = {}
 147         cost = 0.0
 148         for key, fuel in fuels.items():
 149             fresh = [plan.burns[plant][key].fresh_m3 for plant in plants]
 150             stored = [plan.burns[plant][key].stored_m3 for plant in plants]
 151             for volume in (*fresh, *stored):
 152                 nonnegative(f"day {number} burn", volume)
 153             delivered = plan.trucks[key] * fuel.truck_m3
 154             inventory[key] = InventoryDay(opening[key], delivered, sum(fresh), sum(stored), plan.closing_m3[key])
 155             cost += fuel.dry_tonnes(delivered) * fuel.cost_per_dry_tonne
 156         if sum(plan.closing_m3.values()) > limits.total_storage_m3 + TOL:
 157             raise ValueError(f"day {number}: total storage capacity exceeded")
 158         if limits.pellet_silo_m3 is not None and plan.closing_m3["wp"] > limits.pellet_silo_m3 + TOL:
 159             raise ValueError(f"day {number}: pellet silo capacity exceeded")
 160 
 161         heat = {}
 162         for plant_key, plant in plants.items():
 163             heat[plant_key] = {}
 164             for key, fuel in fuels.items():
 165                 burn = plan.burns[plant_key][key]
 166                 heat[plant_key][key] = fuel.heat(burn.fresh_m3 + burn.stored_m3, plant.efficiency[key])
 167             if sum(heat[plant_key].values()) > limits.plant_heat_capacity_gj[plant_key] + TOL:
 168                 raise ValueError(f"day {number}: {plant_key} heat capacity exceeded")
 169         nonnegative(f"day {number} gas heat", plan.gas_gj)
 170         if limits.gas_capacity_gj is not None and plan.gas_gj > limits.gas_capacity_gj + TOL:
 171             raise ValueError(f"day {number}: gas heat capacity exceeded")
 172         if not isclose(sum(sum(values.values()) for values in heat.values()) + plan.gas_gj,
 173                        limits.demand_gj, rel_tol=0, abs_tol=TOL):
 174             raise ValueError(f"day {number}: useful heat does not meet demand")
 175         cost += plan.gas_gj * scenario.gas_cost_per_useful_gj
 176         emissions = daily_emissions_tco2e(
 177             ((heat[p][f], plants[p].emissions_per_useful_gj[f]) for p in plants for f in fuels),
 178             gas_heat_gj=plan.gas_gj, gas_factor_tco2e_per_gj=scenario.gas_emissions_per_useful_gj,
 179         )
 180         audited.append(DayAudit(inventory, heat, cost, emissions))
 181         opening = dict(plan.closing_m3)
 182 
 183     for key in fuels:
 184         terminal = None if scenario.terminal_m3 is None else scenario.terminal_m3[key]
 185         try:
 186             validate_inventory((day.inventory_by_fuel[key] for day in audited),
 187                                initial=scenario.initial_m3[key], terminal=terminal, tolerance=TOL)
 188         except ValueError as error:
 189             raise ValueError(f"{key}: {error}") from error
 190     return tuple(audited)

 193 def solve(scenario: Scenario) -> SolveResult:
 194     fuels, plants = scenario.feedstocks, scenario.plants
 195     columns = {}
 196     costs, integrality, variable_upper = [], [], []
 197     row_indices, column_indices, values, lower, upper = [], [], [], [], []
 198 
 199     def variable(key: tuple, cost: float = 0, integer: bool = False, maximum: float = inf) -> None:
 200         columns[key] = len(costs)
 201         costs.append(cost)
 202         integrality.append(int(integer))
 203         variable_upper.append(maximum)
 204 
 205     def row(terms: dict[tuple, float], lo: float = -inf, hi: float = inf) -> None:
 206         for key, coefficient in terms.items():
 207             if coefficient != 0:
 208                 row_indices.append(len(lower))
 209                 column_indices.append(columns[key])
 210                 values.append(coefficient)
 211         lower.append(lo)
 212         upper.append(hi)
 213 
 214     for t, day in enumerate(scenario.days):
 215         variable(("gas", t), scenario.gas_cost_per_useful_gj,
 216                  maximum=inf if day.gas_capacity_gj is None else day.gas_capacity_gj)
 217         for f, fuel in fuels.items():
 218             variable(("trucks", t, f), fuel.dry_tonnes(fuel.truck_m3) * fuel.cost_per_dry_tonne, integer=True)
 219             variable(("stock", t, f))
 220             for p in plants:
 221                 variable(("fresh", t, f, p))
 222                 variable(("stored", t, f, p))
 223 
 224     for t, day in enumerate(scenario.days):
 225         row({("trucks", t, f): 1 for f in fuels}, hi=day.truck_limit)
 226         row({("stock", t, f): 1 for f in fuels}, hi=day.total_storage_m3)
 227         if day.pellet_silo_m3 is not None:
 228             row({("stock", t, "wp"): 1}, hi=day.pellet_silo_m3)
 229         for f, fuel in fuels.items():
 230             fresh = {("fresh", t, f, p): 1 for p in plants}
 231             stored = {("stored", t, f, p): 1 for p in plants}
 232             delivery = {("trucks", t, f): -fuel.truck_m3}
 233             prior = {("stock", t - 1, f): -1} if t else {}
 234             initial = 0 if t else scenario.initial_m3[f]
 235             row({**fresh, **delivery}, hi=0)
 236             row({**stored, **prior}, hi=initial)
 237             row({("stock", t, f): 1, **fresh, **stored, **delivery, **prior}, lo=initial, hi=initial)
 238 
 239         demand = {("gas", t): 1.0}
 240         for p, plant in plants.items():
 241             heat = {(stream, t, f, p): fuel.heat(1, plant.efficiency[f])
 242                     for f, fuel in fuels.items() for stream in ("fresh", "stored")}
 243             row(heat, hi=day.plant_heat_capacity_gj[p])
 244             demand.update(heat)
 245         row(demand, lo=day.demand_gj, hi=day.demand_gj)
 246 
 247     if scenario.terminal_m3 is not None:
 248         for f, value in scenario.terminal_m3.items():
 249             row({("stock", len(scenario.days) - 1, f): 1}, lo=value, hi=value)
 250 
 251     matrix = coo_array((values, (row_indices, column_indices)), shape=(len(lower), len(costs))).tocsc()
 252     answer = milp(
 253         costs, integrality=integrality, bounds=Bounds(np.zeros(len(costs)), variable_upper),
 254         constraints=LinearConstraint(matrix, lower, upper), options={"mip_rel_gap": 0.0},
 255     )
 256     bound, gap = getattr(answer, "mip_dual_bound", None), getattr(answer, "mip_gap", None)
 257     if answer.status != 0:
 258         return SolveResult(answer.status, answer.message, bound, gap)
 259 
 260     def value(*key: str | int) -> float:
 261         return float(answer.x[columns[key]])
 262 
 263     plans = []
 264     for t in range(len(scenario.days)):
 265         trucks = {}
 266         for f in fuels:
 267             count = value("trucks", t, f)
 268             trucks[f] = round(count)
 269             if not isclose(count, trucks[f], rel_tol=0, abs_tol=TOL):
 270                 raise ValueError("solver returned a fractional truck count")
 271         plans.append(DayPlan(
 272             trucks, {f: value("stock", t, f) for f in fuels},
 273             {p: {f: Burn(value("fresh", t, f, p), value("stored", t, f, p)) for f in fuels} for p in plants},
 274             value("gas", t),
 275         ))
 276     audit = verify_plan(scenario, plans)
 277     cost = sum(day.cost for day in audit)
 278     if (not isclose(cost, answer.fun, rel_tol=0, abs_tol=TOL)
 279             or bound is None or not isclose(cost, bound, rel_tol=0, abs_tol=TOL)
 280             or gap is None or not isclose(gap, 0, rel_tol=0, abs_tol=TOL)):
 281         raise ValueError("solver objective, bound or gap disagrees with the independently checked cost")
 282     return SolveResult(answer.status, answer.message, bound, gap, tuple(plans), audit)

 285 def synthetic_result(scenario: Scenario, output: SolveResult) -> dict:
 286     """Package the synthetic example for JSON export without writing a file."""
 287     return {
 288         "schema_version": 1,
 289         "data_kind": "synthetic_test_fixture",
 290         "scenario": scenario.name,
 291         "feedstocks": list(scenario.feedstocks),
 292         "plants": list(scenario.plants),
 293         "units": {"volume": "m3", "heat": "GJ useful", "cost": "synthetic USD", "emissions": "tCO2e"},
 294         "solver": {"backend": "HiGHS via SciPy milp", "scipy_version": scipy.__version__,
 295                    "status": output.status, "message": output.message,
 296                    "best_cost_bound": output.best_bound, "mip_gap": output.mip_gap},
 297         "totals": {"cost": sum(day.cost for day in output.audit),
 298                    "emissions_tco2e": sum(day.emissions_tco2e for day in output.audit)} if output.audit else None,
 299         "days": [
 300             {"day": index, "limits": asdict(limits), **asdict(plan), **asdict(audit)}
 301             for index, (limits, plan, audit) in enumerate(
 302                 zip(scenario.days[:len(output.plans)], output.plans, output.audit, strict=True), start=1
 303             )
