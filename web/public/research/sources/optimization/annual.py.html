  23 def solver_version() -> dict:
  24     # The public milp result gives no HiGHS version. Read the pinned wheel's
  25     # bundled constants if exposed, without depending on them to solve.
  26     try:
  27         from scipy.optimize._highspy import _core
  28         version = ".".join(str(getattr(_core, f"HIGHS_VERSION_{part}")) for part in ("MAJOR", "MINOR", "PATCH"))
  29     except (ImportError, AttributeError):
  30         version = None
  31     return {"backend": "HiGHS via SciPy milp", "scipy_version": scipy.__version__, "highs_version": version}

  34 def annual_audit(scenario: phase3.Scenario, plans: tuple[phase3.DayPlan, ...], objective: float) -> tuple[tuple, dict]:
  35     """Recheck all daily constraints and reconcile whole-horizon totals."""
  36     audited = phase3.verify_plan(scenario, plans)
  37     residuals = {}
  38     purchase_costs = []
  39     for f, fuel in scenario.feedstocks.items():
  40         ledger = [day.inventory_by_fuel[f] for day in audited]
  41         delivered = fsum(item.delivered for item in ledger)
  42         burned = fsum(item.fresh_burn + item.stored_burn for item in ledger)
  43         residuals[f] = fsum((scenario.initial_m3[f], delivered, -burned, -ledger[-1].closing))
  44         if abs(residuals[f]) > FLOW_TOL:
  45             raise ValueError(f"{f}: annual material balance residual exceeds tolerance")
  46         purchase_costs.append(delivered * fuel.wet_density * (1 - fuel.moisture) * fuel.cost_per_dry_tonne)
  47     gas_gj = fsum(plan.gas_gj for plan in plans)
  48     biomass_gj = fsum(
  49         heat for day in audited for plant in day.biomass_gj_by_plant_and_fuel.values() for heat in plant.values()
  50     )
  51     demand = fsum(day.demand_gj for day in scenario.days)
  52     if not isclose(biomass_gj + gas_gj, demand, rel_tol=0, abs_tol=FLOW_TOL):
  53         raise ValueError("annual useful heat does not equal demand")
  54     cost = fsum(purchase_costs) + gas_gj * scenario.gas_cost_per_useful_gj
  55     if not all(isclose(cost, value, rel_tol=0, abs_tol=COST_TOL)
  56                for value in (objective, fsum(day.cost for day in audited))):
  57         raise ValueError("annual recomputed cost differs from solver objective")
  58     return audited, {
  59         "cost": cost, "demand_gj": demand, "gas_gj": gas_gj, "biomass_gj": biomass_gj,
  60         "emissions_tco2e": fsum(day.emissions_tco2e for day in audited),
  61         "fuel_balance_residual_m3": residuals,
  62     }

  65 def finite(value, scale: float = 1) -> float | None:
  66     if value is None or not isfinite(value):
  67         return None
  68     return float(value) * scale

  71 def decode_incumbent(scenario: phase3.Scenario, vector) -> tuple[phase3.DayPlan, ...]:
  72     """Read the unchanged Phase 3 column order when it returns no plan."""
  73     width = 1 + len(scenario.feedstocks) * (2 + 2 * len(scenario.plants))
  74     plans = []
  75     for values in vector.reshape(len(scenario.days), width):
  76         entries = iter(float(value) for value in values)
  77         gas = next(entries)
  78         trucks, closing = {}, {}
  79         burns = {plant: {} for plant in scenario.plants}
  80         for fuel in scenario.feedstocks:
  81             count = next(entries)
  82             trucks[fuel] = round(count)
  83             if not isclose(count, trucks[fuel], rel_tol=0, abs_tol=phase3.TOL):
  84                 raise ValueError("solver incumbent has a fractional truckload")
  85             closing[fuel] = next(entries)
  86             for plant in scenario.plants:
  87                 burns[plant][fuel] = phase3.Burn(next(entries), next(entries))
  88         plans.append(phase3.DayPlan(trucks, closing, burns, gas))
  89     return tuple(plans)

  92 def run(loaded: LoadedScenario, time_limit: float) -> dict:
  93     """Run one experiment synchronously. The CLI uses a dedicated process."""
  94     if not isfinite(time_limit) or time_limit <= 0:
  95         raise ValueError("solver time limit must be finite and positive")
  96     original_milp = phase3.milp
  97     raw = None
  98     seconds = None
  99 
 100     def bounded_milp(*args, **kwargs):
 101         nonlocal raw, seconds
 102         options = {**kwargs.pop("options", {}), "time_limit": time_limit}
 103         started = perf_counter()
 104         raw = original_milp(*args, options=options, **kwargs)
 105         seconds = perf_counter() - started
 106         if raw.x is not None:
 107             # HiGHS can return tiny negative values at zero. Keep material
 108             # violations visible and canonicalize only within Phase 3 tolerance.
 109             raw.x = raw.x.copy()
 110             raw.x[(raw.x < 0) & (raw.x >= -phase3.TOL)] = 0.0
 111         return raw
 112 
 113     model = loaded.scenario
 114     scaled = replace(model, gas_cost_per_useful_gj=model.gas_cost_per_useful_gj / COST_SCALE,
 115                      feedstocks={key: replace(fuel, cost_per_dry_tonne=fuel.cost_per_dry_tonne / COST_SCALE)
 116                                  for key, fuel in model.feedstocks.items()})
 117     # Phase 3 remains byte-for-byte unchanged. In the isolated experiment
 118     # process, add a solver deadline and retain raw termination diagnostics.
 119     validation_error = None
 120     with patch.object(phase3, "milp", bounded_milp):
 121         try:
 122             result = phase3.solve(scaled)
 123         except ValueError as error:
 124             if raw is None or raw.status != 0:
 125                 raise
 126             validation_error = str(error)
 127             result = None
 128 
 129     objective = finite(getattr(raw, "fun", None), COST_SCALE)
 130     bound = finite(getattr(raw, "mip_dual_bound", None), COST_SCALE)
 131     diagnostics = {
 132         **solver_version(), "status_code": int(raw.status), "message": raw.message,
 133         "runtime_seconds": seconds, "time_limit_seconds": time_limit,
 134         "solver_reported_objective": objective, "best_bound": bound, "mip_gap": finite(getattr(raw, "mip_gap", None)),
 135     }
 136     record = {
 137         "schema_version": 1, "data_kind": "historical_input_experiment", "campus_inputs_validated": False,
 138         "emissions_accuracy": "unverified_factors_and_heat_basis",
 139         "scenario_id": model.name, "label": loaded.label, "days_requested": len(model.days),
 140         "input_sha256": loaded.input_sha256, "assumptions": list(loaded.assumptions),
 141         "units": {"cost": "source dollars", "heat": "GJ useful", "volume": "m3", "emissions": "tCO2e provisional"},
 142         "inputs": asdict(model), "solver": diagnostics, "validation": "no_verified_plan", "totals": None, "days": [],
 143     }
 144     if raw.status not in (0, 1) or raw.x is None or objective is None:
 145         return record
 146     try:
 147         if validation_error is not None:
 148             raise ValueError(validation_error)
 149         decoded = decode_incumbent(model, raw.x)
 150         if raw.status == 0 and decoded != result.plans:
 151             raise ValueError("annual decoder disagrees with the Phase 3 column order")
 152         audited, totals = annual_audit(model, decoded, objective)
 153         if raw.status == 0 and (bound is None or not isclose(totals["cost"], bound, rel_tol=0, abs_tol=COST_TOL)):
 154             raise ValueError("annual objective does not match the solver bound")
 155         if bound is not None and bound > totals["cost"] + COST_TOL:
 156             raise ValueError("solver bound exceeds independently checked incumbent cost")
 157     except ValueError as error:
 158         record.update(validation="audit_failed", validation_error=str(error))
 159         return record
 160     record.update(
 161         validation="verified_against_stated_scenario" if raw.status == 0 else "audited_feasible_incumbent",
 162         totals=totals,
 163         days=[{"day": number, "demand_gj": limits.demand_gj, **asdict(plan), **asdict(audit)}
 164               for number, (plan, audit, limits) in enumerate(zip(decoded, audited, model.days, strict=True), start=1)],
 165     )
 166     return record

 169 def export(record: dict, directory: Path) -> None:
 170     directory.mkdir(parents=True, exist_ok=True)
 171     (directory / "result.json").write_text(json.dumps(record, indent=2, allow_nan=False) + "\n", encoding="utf-8")
 172     with (directory / "flows.csv").open("w", newline="", encoding="utf-8") as handle:
 173         writer = csv.writer(handle)
 174         writer.writerow(("day", "plant", "feedstock", "trucks", "opening_m3", "delivered_m3", "closing_m3",
 175                          "fresh_burn_m3", "stored_burn_m3", "useful_heat_gj", "day_demand_gj", "day_cost", "day_emissions_tco2e"))
 176         for day in record["days"]:
 177             for fuel, inventory in day["inventory_by_fuel"].items():
 178                 writer.writerow((day["day"], "", fuel, day["trucks"][fuel], inventory["opening"],
 179                                  inventory["delivered"], inventory["closing"], "", "", "", "", "", ""))
 180             for plant, fuels in day["burns"].items():
 181                 for fuel, burn in fuels.items():
 182                     writer.writerow((day["day"], plant, fuel, "", "", "", "", burn["fresh_m3"],
 183                                      burn["stored_m3"], day["biomass_gj_by_plant_and_fuel"][plant][fuel], "", "", ""))
 184             writer.writerow((day["day"], "gas", "", "", "", "", "", "", "", day["gas_gj"],
 185                              day["demand_gj"], day["cost"], day["emissions_tco2e"]))

 188 def main() -> int:
 189     parser = argparse.ArgumentParser(description=__doc__)
 190     parser.add_argument("--config", type=Path, default=DEFAULT_CONFIG)
 191     parser.add_argument("--scenario", default="gams-historical-inputs")
 192     parser.add_argument("--days", type=int)
 193     parser.add_argument("--time-limit", type=float, default=30)
 194     parser.add_argument("--output", type=Path)
 195     args = parser.parse_args()
 196     record = run(load_scenario(args.config, args.scenario, args.days), args.time_limit)
 197     if args.output:
 198         export(record, args.output)
 199     summary_keys = ("scenario_id", "days_requested", "input_sha256", "validation", "solver", "totals", "validation_error")
 200     print(json.dumps({key: record[key] for key in summary_keys if key in record}, allow_nan=False))
 201     return 0 if record["validation"] == "verified_against_stated_scenario" else 2
