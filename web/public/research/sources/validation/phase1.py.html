   8 @dataclass(frozen=True)
   9 class InventoryDay:
  10     """Volumes in m³ for one feedstock and one day."""
  11 
  12     opening: float
  13     delivered: float
  14     fresh_burn: float
  15     stored_burn: float
  16     closing: float
  17

  19 def nonnegative(name: str, value: float) -> None:
  20     if not isfinite(value) or value < 0:
  21         raise ValueError(f"{name} must be finite and nonnegative")

  24 def validate_inventory(
  25     days: Iterable[InventoryDay], *, initial: float,
  26     terminal: float | None = None, tolerance: float = 1e-9,
  27 ) -> None:
  28     """Check one feedstock's ledger in day order, with tolerance in m³."""
  29     nonnegative("initial inventory", initial)
  30     nonnegative("tolerance", tolerance)
  31     if terminal is not None:
  32         nonnegative("terminal inventory", terminal)
  33 
  34     previous = initial
  35     for number, day in enumerate(days, start=1):
  36         for name, value in vars(day).items():
  37             nonnegative(f"day {number} {name}", value)
  38         if not isclose(day.opening, previous, rel_tol=0, abs_tol=tolerance):
  39             raise ValueError(f"day {number}: opening inventory differs from prior closing")
  40         if day.fresh_burn > day.delivered + tolerance:
  41             raise ValueError(f"day {number}: fresh burn exceeds deliveries")
  42         if day.stored_burn > day.opening + tolerance:
  43             raise ValueError(f"day {number}: stored burn exceeds opening inventory")
  44         expected = day.opening + day.delivered - day.fresh_burn - day.stored_burn
  45         if not isclose(day.closing, expected, rel_tol=0, abs_tol=tolerance):
  46             raise ValueError(f"day {number}: closing inventory does not conserve volume")
  47         previous = day.closing
  48 
  49     if terminal is not None and not isclose(previous, terminal, rel_tol=0, abs_tol=tolerance):
  50         raise ValueError("closing inventory differs from the terminal target")

  53 def useful_heat_gj(
  54     volume_m3: float, wet_density_t_per_m3: float, moisture_fraction: float,
  55     heating_value_gj_per_dry_t: float, efficiency: float,
  56 ) -> float:
  57     """Convert delivered or stored volume to useful heat in GJ."""
  58     nonnegative("volume", volume_m3)
  59     if not isfinite(wet_density_t_per_m3) or wet_density_t_per_m3 <= 0:
  60         raise ValueError("wet density must be finite and positive")
  61     if not isfinite(heating_value_gj_per_dry_t) or heating_value_gj_per_dry_t <= 0:
  62         raise ValueError("dry heating value must be finite and positive")
  63     if not 0 <= moisture_fraction < 1:
  64         raise ValueError("wet-basis moisture fraction must be in [0, 1)")
  65     if not 0 <= efficiency <= 1:
  66         raise ValueError("efficiency must be in [0, 1]")
  67     dry_tonnes = volume_m3 * wet_density_t_per_m3 * (1 - moisture_fraction)
  68     return dry_tonnes * heating_value_gj_per_dry_t * efficiency

  71 def daily_emissions_tco2e(
  72     biomass_heat_and_factors: Iterable[tuple[float, float]],
  73     *, gas_heat_gj: float, gas_factor_tco2e_per_gj: float,
  74 ) -> float:
  75     """Use factors expressed per GJ of useful heat and count gas once."""
  76     nonnegative("gas heat", gas_heat_gj)
  77     nonnegative("gas emissions factor", gas_factor_tco2e_per_gj)
  78     biomass_emissions = 0.0
  79     for heat, factor in biomass_heat_and_factors:
  80         nonnegative("biomass heat", heat)
  81         nonnegative("biomass emissions factor", factor)
  82         biomass_emissions += heat * factor
  83     return biomass_emissions + gas_heat_gj * gas_factor_tco2e_per_gj
